cmake_minimum_required(VERSION 4.4)
project(stana_consumer LANGUAGES CXX)
if(DEFINED STANA_SOURCE)
	add_subdirectory("${STANA_SOURCE}" stana-source)
else()
	find_package(stana 0.1 CONFIG REQUIRED)
endif()
add_executable(consumer main.cpp)
target_link_libraries(consumer PRIVATE stana::stana)
set_target_properties(consumer PROPERTIES CXX_STANDARD 23 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
enable_testing()
add_test(NAME consumer.run COMMAND consumer)

add_executable(rejected-old-canvas EXCLUDE_FROM_ALL old_canvas.cpp)
target_link_libraries(rejected-old-canvas PRIVATE stana::stana)
set_target_properties(rejected-old-canvas PROPERTIES CXX_STANDARD 23 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)

foreach(fixture IN ITEMS native_shader native_library raw_points reactive_mismatch asset_point asset_transform frame_clock)
	add_executable(rejected-${fixture} EXCLUDE_FROM_ALL "${fixture}.cpp")
	target_link_libraries(rejected-${fixture} PRIVATE stana::stana)
	set_target_properties(rejected-${fixture} PROPERTIES CXX_STANDARD 23 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
endforeach()
